\[
\frac{a^2}{b^2} + 1 = 6 \cdot \frac{a}{b}
\]

\[
\left(\frac{a}{b}\right)^2 - 6\left(\frac{a}{b}\right) + 1 = 0
\]

\[
x^2 - 6x + 1 = 0
\]

\[

\]

\[

  \]

  \]

  \]
