\[ n^{n+1} = n^n \cdot n = m \cdot n \]

\[ n^{n^{n+1}} = n^{m \cdot n} \]

\[ m \cdot n = n^n \cdot n = n^{n + 1} \]

\[ n^{n^{n+1}} = (n^n)^{n+1} = m^{n+1} \]

\[ n^{n(n+1)} = n^{m} \]

\[ n^{n^{n+1}} = (n^n)^{n+1} = m^{n+1} \]
