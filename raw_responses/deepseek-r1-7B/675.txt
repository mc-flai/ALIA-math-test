   \[
   \frac{1}{8} \times 24 = 3 \text{ gossos}
   \]

   \[
   \frac{1}{4} \times 24 = 6 \text{ vaeques}

   \frac{1}{3} \times 24 = 8 \text{ gats}
   \]
