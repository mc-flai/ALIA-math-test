x² + x − 7 = 0

x = (-b ± √(b² - 4ac)) / 2a

x = (-(1) ± √((1)² - 4(1)(-7))) / 2(1)

x = (-1 ± √29) / 2

a = (-1 + √29) / 2
b = (-1 - √29) / 2
