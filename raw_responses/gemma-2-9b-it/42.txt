   x = (x - 1) + (x - 2) + (x - 3) + (x - 3) + 4

   x = 4x - 9

   a + b + c + d = 3 + (3 - 1) + (3 - 2) + (3 - 3) = 3 + 2 + 1 + 0 = 6
